# text-hash.sh:
#!/bin/bash
# Builds release rexx-run from each named archived tree, moved in turn to one
# path, in its own target dir, and prints the sha256 of its .text section.
set -e
S=/tmp/claude-1000/p6-t23
for t in "$@"; do
    mv $S/$t $S/tree
    find $S/tree -type f -exec touch {} +
    ( cd $S/tree/rust && CARGO_TARGET_DIR=$S/target-$t cargo build --release --bin rexx-run 2>&1 | grep -E 'Compiling rexx-exec|Finished|error' )
    mv $S/tree $S/$t
    objcopy -O binary --only-section=.text $S/target-$t/release/rexx-run $S/$t.text
    echo "$t $(sha256sum < $S/$t.text | cut -d' ' -f1) $(stat -c %s $S/$t.text)"
done
