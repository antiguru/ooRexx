# Phase 7 corpus subset -- streams and platform. The union with the earlier
# phase-*.txt files is what the harnesses read, not this file alone.
#
# One path per line, relative to this directory (rust/corpus/). Lines starting
# with # are comments; blank lines are ignored.
#
# A program here may create, read and delete files inside its own run
# directory, which the earlier phases' determinism rule forbade outright. It
# prints no timestamp, no size of a file it did not write, no directory listing
# it did not create, and no host variable's value.

# Task 3: VALUE's two external pools, and QUALIFY, USERID, SETLOCAL, ENDLOCAL
# against the interpreter's own environment and current directory.
lang/platform_value_environment.rex
lang/platform_qualify_setlocal.rex

# Task 3: the REXX package's own three routines, which resolve after a
# program's ::ROUTINEs and before the external file search.
lang/platform_internal_routines.rex

# Task 4: everything a Stream answers before anything is opened.
lang/stream_unopened_queries.rex

# Task 5: the OPEN option table -- the modes that open, and the shapes that
# are a bare 93 or an ERROR:n.
lang/stream_open_modes.rex
lang/stream_open_errors.rex

# Task 6: reading and writing characters and lines, where a line ends, where a
# write lands, and the line count the oracle caches.
lang/stream_chars_lines.rex
lang/stream_line_ends.rex
lang/stream_write_positions.rex

# Task 7: NOTREADY -- which operations raise it, what a handler reads back, and
# the two delivery shapes.
lang/stream_notready_signal.rex
lang/stream_notready_trap.rex

# Task 8: QUERY STREAMTYPE answers from the open descriptor, not the name.
lang/stream_streamtype.rex
lang/stream_seek.rex
lang/stream_seek_line.rex
lang/stream_seek_errors.rex
lang/stream_lineout_position.rex
lang/stream_builtins.rex
lang/stream_table_sharing.rex

# Task 10: the three standard streams as objects -- their names and state, the
# buffer they share with SAY, and the counters an undrained empty input answers.
lang/std_stream_writes.rex
lang/local_std_objects.rex

# The standard streams take no part in finalization, so a finalizer still
# reaches every route after `.local` holds them.
lang/uninit_after_std_streams.rex

# A stream write requires a string value: `makeString` decides what an object
# writes, and an array writes its items joined by a newline.
lang/stream_write_conversion.rex

# Task 11: every trace line reaches `.TRACEOUTPUT` as one `LINEOUT` of a
# `TraceObject`, redirected by the monitor's own destination.
lang/trace_redirection.rex

# Task 11: SAY sends SAY to whatever `.local` holds under OUTPUT, and a
# redirected monitor destination receives it.
lang/output_redirection.rex

# Task 11: a read with no stream name is a message to `.INPUT` -- `LINEIN` and
# `CHARIN` with no arguments, `LINES` with `NORMAL`, and `PULL` as a `LINEIN`.
lang/input_redirection.rex

# Task 12: the array and supplier group -- `arrayIn` fills an array the Rexx
# half allocated and ends on the end-of-input raise, and `StreamSupplier`
# walks the lines.
lang/stream_array.rex

# Task 13: `.File` -- the predicates, the listing, a timestamp round trip, and
# the mutation answers, over a tree the program builds in its own directory.
lang/file_queries.rex

# Task 14: a command's return code -- `RC`, `.RS`, the `ERROR` and `FAILURE`
# traps, and the reraise that lets an error handler catch an untrapped
# failure. Its `sh -c 'exit 127'` clauses also carry the `+++ "RC(n)"` line,
# which the default `TRACE N` emits with no `TRACE` instruction in the file.
lang/command_dispatch.rex

# Task 14: which handler an `ADDRESS` name selects, `PATH` splitting a command
# itself, the one-off form leaving the activation's pair untouched, and an
# environment nothing is registered for.
lang/command_environments.rex

# Task 14: `cd`, `set`, `unset` and `export` run against the interpreter's own
# directory and environment rather than in a child, the metacharacter that
# sends one to a shell instead, and the child that inherits both shadows.
lang/command_internal.rex

# Task 15: `::OPTIONS ERROR SYNTAX` turns a command's untrapped ERROR into
# 98.970. The escalation happens where the condition is raised, so the failure
# retrace never runs -- which is what parts this from the row below.
lang/command_options_error_syntax.rex

# Task 15: the same for FAILURE and 98.971, whose wording differs and which
# beats the rename an untrapped FAILURE would otherwise get. Under the default
# `TRACE N` this program would carry a `+++ "RC(127)"` block if the escalation
# were checked after the trace rather than before it.
lang/command_options_failure_syntax.rex

# Task 15: the condition object `CONDITION('O')` answers. Which indexes it
# carries depends on the condition, so this covers a command's (with `RESULT`)
# and a raised `USER`'s (with `ADDITIONAL`), both through `CALL ON` traps that
# resume so one file shows two kinds.
lang/condition_object.rex

# Task 16: `ADDRESS ... WITH` and a stem. Where the output lands, what
# `REPLACE` keeps and `APPEND` continues from, where the captured bytes are cut
# into lines, and what a stem feeds a command's standard input.
lang/address_with_stem.rex

# Task 16: `USING`, whose object decides what it means, and the merge between
# the configuration stored under an environment name and the one an
# instruction carries -- per stream, with `NORMAL` resetting one. Ends with
# the four errors a redirection raises, each re-arming the trap.
lang/address_with_using.rex

# Task 15: a `SYNTAX` condition's own five extra indexes -- `CODE`,
# `ERRORTEXT`, `MESSAGE`, `ADDITIONAL` and `RC` -- in a file of its own,
# because `CALL ON SYNTAX` is a 25.1 translation error and the kind cannot be
# trapped resumably.
lang/condition_object_syntax.rex
