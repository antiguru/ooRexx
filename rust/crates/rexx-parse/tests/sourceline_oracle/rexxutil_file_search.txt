show: procedure expose s.
  do i = 1 to s.0
    say '  ' i c2x(s.i)
  end
  return

make: procedure
  parse arg name
  call lineout name, 'x'
  call lineout name
  return

t: procedure expose f b
  parse arg code
  signal on syntax name bad
  interpret code
  say 'ok   ' code
  return
bad:
  c = condition('O')
  say 'error' translate(code, '?', '00'x) '|' c~code c~message
  return
