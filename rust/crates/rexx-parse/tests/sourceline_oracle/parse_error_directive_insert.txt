show: procedure
  parse arg src
  signal on syntax
  p = .package~new('x', .array~of(src))
  say src '-> no error'
  return
syntax:
  c = condition('o')
  a = c~additional
  line = src '->' c~code 'items' a~items
  do i = 1 to a~items
    line = line '['a[i]']'
  end
  say line
  say '   ' c~message
  return
