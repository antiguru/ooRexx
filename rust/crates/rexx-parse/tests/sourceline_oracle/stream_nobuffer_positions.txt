/* Where a write lands after a write placed inside the oracle's read buffer.
   A line-positioned LINEOUT, or a write after a line SEEK, reports its write
   position from the end of the buffered read, while its bytes and the next
   unplaced LINEOUT or CHAROUT land after what it wrote. A read, CHARS, LINES
   or an explicit position ends that, and the next write lands where the
   position says. A relative SEEK moves from the position the oracle reports.
   Binary record streams place every write where it is reported. */

scenario:
  f = 'f' || arg(1)~changestr(' ', '_')~changestr(',', '') || '.txt'
  call charout f, 'line1' || '0a'x || 'line2' || '0a'x || 'line3' || '0a'x
  call charout f
  call stream f, 'c', 'open both'
  call run arg(1), arg(2), arg(3), arg(4)
  return

binary:
  f = 'b' || arg(1)~changestr(' ', '_')~changestr(',', '') || '.bin'
  call charout f, 'AAAABBBBCCCCDDDD'
  call charout f
  call stream f, 'c', 'open both binary reclength 4'
  call run arg(1), arg(2), arg(3), arg(4)
  return

run:
  out = arg(1) ':'
  do i = 2 to arg()
    if arg(i) == '' then iterate
    parse value arg(i) with op text
    select
      when op = 'L' then r = lineout(f, text)
      when op~left(1) = 'L' then r = lineout(f, text, op~substr(2))
      when op = 'C' then r = charout(f, text)
      when op~left(1) = 'C' then r = charout(f, text, op~substr(2))
      when op = 'R' then r = linein(f)
      when op = 'I' then r = charin(f, , 2)
      when op = 'N' then r = chars(f)
      when op = 'M' then r = lines(f)
      when op = 'S' then r = stream(f, 'c', 'seek' text)
    end
    out = out op'='r 'w' || stream(f, 'c', 'query position write') ,
      'r' || stream(f, 'c', 'query position read')
  end
  call stream f, 'c', 'close'
  say out
  say '   [' || charin(f, 1, 60)~changestr('00'x, '@')~changestr('0a'x, '|') || ']'
  call stream f, 'c', 'close'
  return
