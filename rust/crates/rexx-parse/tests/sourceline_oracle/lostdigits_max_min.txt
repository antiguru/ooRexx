t: procedure
  parse arg expr, quiet
  signal on lostdigits
  numeric digits 9
  interpret 'x=' expr
  if quiet == '' then say left(expr, 30) 'none' x
  else say left(expr, 30) 'none'
  return
lostdigits:
  say left(expr, 30) condition('C') '['condition('D')']' sigl
  return
