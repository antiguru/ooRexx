::CLASS Ranked
::METHOD init
  expose r
  use arg r
::METHOD rank
  expose r
  return r
::METHOD compareTo
  expose r
  use arg other
  return r - other~rank

::CLASS Broken
::METHOD compareTo
  return 'zzz'

::CLASS Tracer SUBCLASS Comparator
::METHOD compare
  use arg l, r
  say '  cmp' l r
  return l - r

::CLASS Quiet SUBCLASS Comparator
::METHOD compare
  expose n
  use arg l, r
  if var('n') then n = n + 1
  else n = 1
  say '  step' n l r
  return l - r

::CLASS Floaty SUBCLASS Comparator
::METHOD compare
  use arg l, r
  if l << r then return '-1.0'
  if l >> r then return '1.0'
  return '0.0'

::CLASS Fixed SUBCLASS Comparator
::METHOD init
  expose v
  use arg v
::METHOD compare
  expose v
  return v
