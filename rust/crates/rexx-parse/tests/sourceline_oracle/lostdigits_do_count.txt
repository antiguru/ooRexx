count 33
/* A DO repetitor or FOR count is tested as a whole number at NUMERIC DIGITS,
   which rounds it, so one with more digits raises LOSTDIGITS first, named
   by its number unrounded. */
counts = "'2.9999999999'#'1111111111.5'#' 3.0000000000 '#'-1.0000000001'#" ||,
  "'1111111111'#'abc'#3#'3.00000000'"
do while counts <> ''
  parse var counts c '#' counts
  call loop 'do' c
  call loop 'do i = 1 for' c
  call loop 'do x over .array~of(1, 2) for' c
  call loop 'do i = 1 to 2 for' c
end
-- Untrapped, the count is rounded and the loop runs.
n = 0
do '2.9999999999'
  n = n + 1
end
say 'untrapped ran' n
exit

loop:
  signal on lostdigits name loop_ld
  signal on syntax name loop_syntax
  ran = 0
  interpret arg(1) '; ran = ran + 1; leave; end'
  say arg(1) '-> ran' ran
  return
loop_ld:
  say arg(1) '-> LOSTDIGITS' '['condition('D')']'
  return
loop_syntax:
  say arg(1) '-> SYNTAX' condition('O')~code
  return
