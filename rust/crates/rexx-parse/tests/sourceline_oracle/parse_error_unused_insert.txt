/* A trapped translation error's ADDITIONAL array, where the oracle passes an
   insert its message text does not use, beside controls that use theirs and
   one that takes none. */
call show 'use arg (a)'
call show 'call ns:"x"'
call show 'call ns:'
call show 'do while 1 until 2; end'
call show 'nop 1'
call show 'x: nop'
call show 'expose a'
exit

show: procedure
  parse arg src
  signal on syntax
  interpret src
  say src '-> no error'
  return
syntax:
  c = condition('o')
  a = c~additional
  line = src '->' c~code 'items' a~items
  do i = 1 to a~items
    line = line '['a[i]']'
  end
  say line
  say '   ' c~message
  return
