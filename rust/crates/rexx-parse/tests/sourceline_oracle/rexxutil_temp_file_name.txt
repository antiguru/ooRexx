-- the answer's properties: its length beyond the qualified template's, the
-- number of places where they differ, which must be filler places that now
-- hold digits, and whether the answer names anything
p: procedure
  parse arg template, filler
  if arg(2, 'e') then name = SysTempFileName(template, filler)
  else do
    name = SysTempFileName(template)
    filler = '?'
  end
  if name == '' then do
    say left(translate(template, '?', '00'x), 28) '[]'
    return
  end
  q = qualify(template)
  changed = 0
  do i = 1 to min(q~length, name~length)
    if q~subchar(i) == name~subchar(i) then iterate
    if q~subchar(i) \== filler | \datatype(name~subchar(i), 'W') then changed = changed + 1000
    changed = changed + 1
  end
  say left(translate(template, '?', '00'x), 28) name~length - q~length changed SysFileExists(name)
  return

e: procedure
  if arg(2, 'e') then say 'empty     ' '['SysTempFileName(arg(1), arg(2))']'
  else say 'empty     ' '['SysTempFileName(arg(1))']'
  return

t: procedure
  parse arg code
  signal on syntax name bad
  interpret code
  say 'ok   ' code
  return
bad:
  c = condition('O')
  say 'error' translate(code, '?', '00'x) '|' c~code translate(c~message, '?', '00'x)
  return
