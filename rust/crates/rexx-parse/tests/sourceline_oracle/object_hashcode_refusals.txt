count 33
/* Object~hashCode takes no arguments, and that is its only refusal: the
 * method itself cannot fail, every receiver having either a value hash or an
 * identity one.
 *
 * Class carries its own row (`AddMethod("HashCode", RexxClass::hashCode, 0)`,
 * memory/Setup.cpp:495) rather than inheriting Object's, so its arity is
 * witnessed separately.
 *
 * The untrapped tail is a class receiver's arity error. rc 163.
 */

signal on syntax name trapped
n = 0

next:
n = n + 1
select
  when n = 1 then say n 'answered' 'abc'~hashCode(1)
  when n = 2 then say n 'answered' .Object~new~hashCode(1)
  when n = 3 then say n 'answered' .String~hashCode(1)
  when n = 4 then say n 'answered' .nil~hashCode(1)
  otherwise signal done
end
signal next

trapped:
say n 'raised' rc'.'condition('E')
signal on syntax name trapped
signal next

done:
signal off syntax
say .String~hashCode(1)
