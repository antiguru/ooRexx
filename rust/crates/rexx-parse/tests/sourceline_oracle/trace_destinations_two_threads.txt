   counter's guard without waiting under a pin for the other's. */

b = .tracer~new~start('run')

::class tracer
::method run
  trace r
  do i = 1 to 30
    x = 1
  end
  trace off
  return 0
::class counting
::method init class
  expose count
  count = 0
::method bump class
  expose count
  count += 1
::method count class
  expose count
  return count
::method lineout
  .counting~bump
  return 0
