/* An INTERPRET fragment's frame keeps one invocation for the fragment's
   life, and lists the fragment's own clause at its own indent; the
   activation running the INTERPRET lists the INTERPRET clause at that
   clause's indent, live and in a trapped condition's traceback. */

    say 'cond' f~type f~line (f~invocation == .nil)
  end
  say 'again' .context~stackframes[1]~invocation .context~invocation
do 1
  interpret 'do 1; do 1; do f over .context~stackframes; say f~traceline; end; end; end'
end

do 1
  call try 'do 1; do 1; x = f(); end; end'
end
