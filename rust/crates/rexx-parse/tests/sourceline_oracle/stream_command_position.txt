/* Stream positions after an implicit open. A read opens read-write and puts
   the write position at the end. QUERY HANDLE answers the descriptor of an
   open stream. A line SEEK on a closed or never opened stream opens it and
   lands. A SEEK offset is an int64_t, so 2**63 is 93. A forward line SEEK
   counts line ends, so it stops on an unterminated last line. */
f = 'p.txt'
call lineout f, '1234567890'
call lineout f
say 'linein [' || linein(f, 1, 0) || ']'
h = stream(f, 'c', 'query handle')
say 'handle' datatype(h, 'W') (h > 2)
say 'position write' stream(f, 'c', 'query position write') ,
  stream(f, 'c', 'query position write line') stream(f, 'c', 'query position read')
say 'close' stream(f, 'c', 'close') '[' || stream(f, 'c', 'query handle') || ']'
say 'open' stream(f, 'c', 'open both')
say 'close' stream(f, 'c', 'close')
say 'seek closed' stream(f, 'c', 'seek =1 line')
say 'close' stream(f, 'c', 'close')
s = .Stream~new(f)
say 'seek unopened' s~seek('=1 line') s~query('position write')
s~close
