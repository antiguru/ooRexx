   SYNTAX in that file, trapped and then untrapped. */
call try .method, 'badm.rex'
call try .routine, 'badr.rex'
r = .routine~newFile('badu.rex')
say 'not reached'
exit

  say o~code o~position rc o~traceback~items
  do l over o~traceback; say '  tb:' l; end
