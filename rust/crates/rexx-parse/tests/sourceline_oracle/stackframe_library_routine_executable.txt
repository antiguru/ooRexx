::routine show
  use arg tag, condition, expected
  do f over condition~stackFrames
    e = f~executable
    name = f~name
    if f~type == 'PROGRAM' then name = ''
    say tag f~type name e~class~id (e == expected) (e == .routines~r) (e == .routines~q)
  end
