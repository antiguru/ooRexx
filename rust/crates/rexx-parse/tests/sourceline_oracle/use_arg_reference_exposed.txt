try: procedure
  use arg o, message, first, second
  signal on syntax name caught
  if arg() = 3 then answer = o~send(message, first)
  else answer = o~send(message, first, second)
  say message 'answers' answer
  return
caught:
  say message condition('o')~code condition('o')~message
  return

::class k
::method refExposed
  expose abc
  use arg >abc
  return abc
::method refAutoExposed
  use local
  use arg >abc
  return abc
::method refLocal
  use local abc
  use arg >abc
  return abc
::method refExposedStem
  expose abc.
  use arg >abc.
  return abc.1
::method refAutoExposedStem
  use local
  use arg >abc.
  return abc.1
::method refLocalStem
  use local abc.
  use arg >abc.
  return abc.1
::method doubleArgStem
  use arg abc., >abc.
  return abc.1
::method refAfterDrop
  use arg abc.
  drop abc.
  use arg , >abc.
  return abc.1
::method doubleArg
  use arg abc, >abc
  return abc
::method refTwice
  use arg >abc
  use arg >abc
  return abc
::method refTwiceUnset
  use arg >abc
  use arg >abc
  return 'bound twice'
::method kindFirst
  expose abc
  use arg >abc
::method unsetPool
  expose pooled
  use arg >pooled
::method setPool
  expose pooled
  pooled = 'object value'
  use arg >pooled
::method twiceExposed
  expose v
  signal on syntax name caught
  say 'twiceExposed answers' self~bindTwice(>v)
  return
caught:
  say 'twiceExposed' condition('o')~code condition('o')~message
  return
::method bindTwice
  use arg >q
  use arg >q
  return 'bound twice'
