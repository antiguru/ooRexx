t: procedure
  parse arg setup, code
  interpret setup
  signal on syntax name bad
  interpret code
  say 'ok   ' code '|' show()
  return
bad:
  c = condition('O')
  say 'error' code '|' c~code c~message '|' show()
  return

show: procedure expose a.
  o = ''
  do i = 0 to 5; o = o a.i; end
  return o
