/* Which operand receives the `==` when a collection searches for an item.

   `ArrayClass::findSingleIndexItem` is `item->equalValue(test)`: the value
   being SEARCHED FOR receives the message and the slot's contents is its
   argument.  Every line below reverses if the two are swapped, and a witness
   that puts a K in and asks for a K cannot see the difference because both
   directions answer 1. */

k = .Array~of('plain')
say 'holds a string, asked for a K' k~hasItem(.K~new) k~index(.K~new)

j = .Array~new
j~append(.K~new)
say 'holds a K, asked for a string' j~hasItem('plain') j~index('plain')

l = .List~of('plain')
say 'list holds a string, asked for a K' l~hasItem(.K~new) l~index(.K~new)

m = .List~new
zz = m~append(.K~new)
say 'list holds a K, asked for a string' m~hasItem('plain')
