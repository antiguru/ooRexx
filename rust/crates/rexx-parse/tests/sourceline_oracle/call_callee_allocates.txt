count 30
/* A callee that allocates while its caller holds values in registers. */
a = 'left'
do i = 1 to 3
  s = (a || i) || churn(50) || (a || 'x')
  say length(s) left(s, 6)
end
say pair(copies('ab', 3), churn(10))
call pair (a || 'z'), churn(5)
say result
say rt(4) rt(0)
call rt 2
say result
exit
churn: procedure
  arg n
  t = ''
  do j = 1 to n
    t = t || j
    o = .array~new(j)
  end
  return length(t)
pair:
  return arg(1) '/' arg(2)
::routine rt
  if arg(1) = 0 then exit 'none'
  a = .array~new
  do i = 1 to arg(1)
    a~append(copies('r', i))
  end
  return a~items a[a~items]
