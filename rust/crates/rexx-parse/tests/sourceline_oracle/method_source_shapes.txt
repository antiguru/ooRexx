t: procedure
  use arg src
  signal on syntax
  m = .method~new('m', src)
  say "ok" m~source~items "["m~source[1]"]"
  return
syntax: say 'err' condition('o')~code condition('o')~message
  return
::class t
::method rr
  use arg m
  return self~run(m)
