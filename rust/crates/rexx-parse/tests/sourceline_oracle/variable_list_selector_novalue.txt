   NOVALUE: 98.986 under ::OPTIONS NOVALUE SYNTAX, and a raised NOVALUE when
   SIGNAL ON NOVALUE is in force. A set selector raises nothing. A compound
   PROCEDURE EXPOSE selector is read in the procedure's own frame. */

call try o, 'dropUnset'

call try o, 'exposeUnset'
call try o, 'dropSet'

  use arg o, message

::class k
::method dropUnset
  drop (nolist)
  return 'nothing'
::method procedureUnset
  call pe
  return 'nothing'
pe: procedure expose (nolist2)
  return
::method exposeUnset
  expose (nolist3)
  return 'nothing'
::method dropSet
  names = 'v'; v = 1
  drop (names)
  return symbol('v')
::method procedureSet
  names = 'v'; v = 1
  call ps
  return v
ps: procedure expose (names)
  v = 'set in ps'
  return
::method procedureCompound
  i = 2; a.2 = 'v'
  call pk
  return 'nothing'
pk: procedure expose (a.i)
  return
::method procedureConstant
  a.2 = 'v'; v = 1
  call pn
  return v
pn: procedure expose (a.2)
  v = 'set in pn'
  return
::method setNames
  expose names
  names = 'v'
  return names
::method exposeSet
  expose (names)
  v = 'object v'
  return v
::method dropTrapped
  signal on novalue name novalue
  drop (nolist4)
  return 'nothing'
novalue:
  return 'NOVALUE' condition('d') sigl
