::class k
::method dropUnset
  drop (nolist)
  return 'nothing'
::method procedureUnset
  call pe
  return 'nothing'
pe: procedure expose (nolist2)
  return
::method exposeUnset
  expose (nolist3)

::method dropSet
  names = 'v'; v = 1
  drop (names)
  return symbol('v')
::method procedureSet

  call ps
  return v
ps: procedure expose (names)
  v = 'set in ps'
  return
::method setNames

::method exposeSet

  return v
::method dropTrapped
