count 27
/* An UNINIT that ~define gives a superclass after its subclass exists makes
   the subclass's new instances finalizable; deleting it again stops that.
   Collection timing is not specified, so each case drives collections until
   the finalizer has run or a bound is reached, and prints only whether it
   ran. */
first = .Object~subclass('First')
second = first~subclass('Second')
first~define('UNINIT', ".local~ran = .local~ran self~identityHash")
.local~ran = ''
o = second~new
id = o~identityHash
drop o
say 'defined:' collected(id)
first~delete('UNINIT')
o = second~new
id = o~identityHash
drop o
say 'deleted:' collected(id)
exit

collected: procedure
  use arg id
  do 50 while wordpos(id, .local~ran) = 0
    pad = copies('x', 1000)
    call gc 'force'
  end
  return wordpos(id, .local~ran) > 0
